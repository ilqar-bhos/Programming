create table test.departments (dept_id serial primary key, dept_name varchar(55), city varchar(55));

create table test.employees (
emp_id serial primary key,
 name varchar(55),
 age int,
 email varchar(55),
 salary int,
 dept_id int,
 foreign key(dept_id) references test.departments,
 manager_id int,
 foreign key(emp_id) references test.employees,
 hire_date date
);
